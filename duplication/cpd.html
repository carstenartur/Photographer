    private static boolean hasDotSegment(String path) {
        for (String segment : path.split("/", -1)) {
            if (".".equals(segment) || "..".equals(segment)) {
                return true;
            }
        }
        return false;
    }

    private static URI origin(URI uri) {
        String scheme = uri.getScheme();
        String host = uri.getHost();
        if (scheme == null || host == null) {
            throw new IllegalArgumentException("baseUrl must contain a scheme and host");
        }
        try {
            return new URI(
                    scheme.toLowerCase(Locale.ROOT),
                    null,
                    host.toLowerCase(Locale.ROOT),
                    effectivePort(uri),
                    "/",
                    null,
                    null);
        } catch (URISyntaxException exception) {
            throw new IllegalArgumentException("invalid camera base URL", exception);
        }
    }

    private static boolean sameOrigin(URI expected, URI actual) {
        return expected.getScheme().equalsIgnoreCase(actual.getScheme())
                && expected.getHost().equalsIgnoreCase(actual.getHost())
                && effectivePort(expected) == effectivePort(actual);
    }

    private static int effectivePort(URI uri) {
        if (uri.getPort() >= 0) {
            return uri.getPort();
        }
        return "https".equalsIgnoreCase(uri.getScheme()) ? 443 : 80;
    }

private double normalizedCrossCorrelation(float[] a, float[] b) {
        if (a.length != b.length || a.length == 0) return 0.0;

        double meanA = 0, meanB = 0;
        for (int i = 0; i < a.length; i++) {
            meanA += a[i];
            meanB += b[i];
        }
        meanA /= a.length;
        meanB /= b.length;

        double sumAB = 0, sumAA = 0, sumBB = 0;
        for (int i = 0; i < a.length; i++) {
            double da = a[i] - meanA;
            double db = b[i] - meanB;
            sumAB += da * db;
            sumAA += da * da;
            sumBB += db * db;
        }

        double denom = Math.sqrt(sumAA * sumBB);
        if (denom < 1e-10) return 1.0; // both constant → identical
        double ncc = sumAB / denom;
        return Math.max(0.0, Math.min(1.0, ncc));
    }
